-- ===================================================================== -- Dollar Store - المرحلة 3: نظام الاحتجاز (Escrow) + محفظة البائع + سحب أسبوعي (الجمعة) -- شغّله بعد schema.sql ثم schema_phase2.sql. -- آمن لإعادة التشغيل أكتر من مرة. -- -- الفكرة: -- بيع -> بعد التسليم الفلوس "معلّقة" عند البائع (pending) -- بعد فترة الاحتجاز (72 ساعة، أو لحد نهاية الضمان لو أطول) وبدون مشكلة -> "متاحة" (available) -- البائع يطلب سحب -> يدخل دفعة الجمعة القادمة -> الأدمن يحوّل يدويًا ويضغط "تم الدفع" -- الفلوس الفعلية بتفضل في محفظتك أنت، والجداول دي بتسجّل حقوق كل بائع بس. -- ===================================================================== -- --------------------------------------------------------------------- -- 0) إعدادات (أرقام مبدئية، عدّلها بأمر update في آخر الملف) -- --------------------------------------------------------------------- insert into public.app_settings (key, value) values ('hold_hours', '72'), -- مدة الاحتجاز بعد التسليم ('commission_percent', '10'), -- عمولة المنصة على منتجات البائعين (رقم مبدئي) ('min_payout_egp', '100'), -- أقل مبلغ سحب (رقم مبدئي) ('payout_weekday', '5'), -- يوم الدفع: 5 = الجمعة (1 = الاثنين ... 7 = الأحد) ('seller_terms_text', '1) فلوس كل عملية بتتعلّق عند المنصة لحد انتهاء فترة الاحتجاز (72 ساعة من التسليم، أو لحد نهاية الضمان لو أطول).' || E'\n' || '2) لو المشتري بلّغ عن مشكلة، بيتجمّد المبلغ لحد ما الإدارة تفصل فيها.' || E'\n' || '3) بعد انتهاء الفترة بدون مشاكل، المبلغ بيتحول لرصيد متاح تقدر تطلب سحبه.' || E'\n' || '4) طلبات السحب بتتحوّل مرة كل أسبوع يوم الجمعة، وبحد أدنى للسحب بيحدده الموقع.' || E'\n' || '5) عمولة المنصة بتتخصم من كل عملية بيع.' || E'\n' || '6) أي مبلغ اتسلّم للمشتري كاسترجاع بيتخصم من رصيدك المعلّق.' || E'\n' || '7) التوثيق بالهوية شرط قبل أول عملية سحب.') on conflict (key) do nothing; -- الشروط تتقرا للكل (نسخة المشتري ونسخة البائع) drop policy if exists settings_read on public.app_settings; create policy settings_read on public.app_settings for select using (key in ('terms_version', 'terms_text', 'seller_terms_text')); create or replace function public._setting(p_key text, p_default text) returns text language sql stable security definer set search_path = public as $$ select coalesce((select value from public.app_settings where key = p_key), p_default); $$; revoke all on function public._setting(text, text) from public, anon, authenticated; -- --------------------------------------------------------------------- -- 1) ربط المنتجات والطلبات بالبائع -- seller_id = null -> منتج المتجر نفسه (مفيش محفظة بائع) -- --------------------------------------------------------------------- alter table public.products add column if not exists seller_id uuid references public.profiles(id); alter table public.orders add column if not exists seller_id uuid references public.profiles(id), add column if not exists commission_pct numeric(5,2) not null default 0, add column if not exists seller_net numeric(10,2), add column if not exists release_at timestamptz, add column if not exists released_at timestamptz; -- البائع يشوف مبيعاته (من غير بيانات التسليم) drop policy if exists orders_read on public.orders; create policy orders_read on public.orders for select to authenticated using (buyer_id = auth.uid() or seller_id = auth.uid() or public.is_admin()); -- --------------------------------------------------------------------- -- 2) طلبات السحب + دفتر المحفظة (القراءة فقط للعميل) -- --------------------------------------------------------------------- create table if not exists public.payout_requests ( id bigint generated always as identity primary key, seller_id uuid not null references public.profiles(id), amount_egp numeric(10,2) not null check (amount_egp > 0), method text not null check (method in ('vodafone_cash', 'instapay')), account text not null check (char_length(account) between 5 and 60), status text not null default 'requested' check (status in ('requested', 'paid', 'rejected')), batch_date date not null, admin_note text, created_at timestamptz not null default now(), processed_at timestamptz ); alter table public.payout_requests enable row level security; revoke all on public.payout_requests from anon, authenticated; grant select on public.payout_requests to authenticated; drop policy if exists payouts_read on public.payout_requests; create policy payouts_read on public.payout_requests for select to authenticated using (seller_id = auth.uid() or public.is_admin()); create table if not exists public.wallet_entries ( id bigint generated always as identity primary key, seller_id uuid not null references public.profiles(id), order_id bigint references public.orders(id), payout_id bigint references public.payout_requests(id), kind text not null check (kind in ('sale','release','refund_reversal','payout_request','payout_reversal','adjustment')), pending_delta numeric(10,2) not null default 0, available_delta numeric(10,2) not null default 0, note text, created_at timestamptz not null default now() ); create index if not exists wallet_seller_idx on public.wallet_entries (seller_id, created_at desc); alter table public.wallet_entries enable row level security; revoke all on public.wallet_entries from anon, authenticated; grant select on public.wallet_entries to authenticated; drop policy if exists wallet_read on public.wallet_entries; create policy wallet_read on public.wallet_entries for select to authenticated using (seller_id = auth.uid() or public.is_admin()); -- --------------------------------------------------------------------- -- 3) تاريخ دفعة الجمعة القادمة (بتوقيت القاهرة) -- طلب يوم الجمعة نفسه يدخل الجمعة اللي بعدها -- --------------------------------------------------------------------- create or replace function public.next_payout_date(p_ts timestamptz default now()) returns date language plpgsql stable security definer set search_path = public as $$ declare v_day date := (p_ts at time zone 'Africa/Cairo')::date; v_dow int := extract(isodow from v_day)::int; v_w int := public._setting('payout_weekday', '5')::int; begin return v_day + case when v_dow < v_w then v_w - v_dow else v_w - v_dow + 7 end; end $$; revoke all on function public.next_payout_date(timestamptz) from public, anon; grant execute on function public.next_payout_date(timestamptz) to authenticated; -- --------------------------------------------------------------------- -- 4) إنشاء الطلب (نسخة محدّثة: بتسجل البائع والعمولة) -- --------------------------------------------------------------------- create or replace function public.create_order(p_product_id bigint, p_terms_version text) returns public.orders language plpgsql security definer set search_path = public as $$ declare v_uid uuid := auth.uid(); v_prod public.products; v_cur text; v_terms text; v_open int; v_pct numeric(5,2) := 0; v_net numeric(10,2); v_order public.orders; begin if v_uid is null then raise exception 'login_required'; end if; select value into v_cur from public.app_settings where key = 'terms_version'; if p_terms_version is distinct from v_cur then raise exception 'terms_not_accepted'; end if; select * into v_prod from public.products where id = p_product_id and is_active; if not found then raise exception 'product_not_found'; end if; if v_prod.seller_id = v_uid then raise exception 'own_product'; end if; if v_prod.delivery_mode = 'inventory' and not exists (select 1 from public.inventory_items where product_id = v_prod.id and status = 'available') then raise exception 'out_of_stock'; end if; select count(*) into v_open from public.orders where buyer_id = v_uid and status in ('awaiting_payment', 'under_review'); if v_open >= 3 then raise exception 'too_many_open_orders'; end if; if v_prod.seller_id is not null then v_pct := public._setting('commission_percent', '10')::numeric; v_net := round(v_prod.price_egp * (1 - v_pct / 100), 2); end if; select value into v_terms from public.app_settings where key = 'terms_text'; insert into public.orders (buyer_id, product_id, price_egp, warranty_hours, terms_version, terms_snapshot, seller_id, commission_pct, seller_net) values (v_uid, v_prod.id, v_prod.price_egp, v_prod.warranty_hours, v_cur, coalesce(v_terms, '') || E'\n---\n' || coalesce(v_prod.warranty_terms, ''), v_prod.seller_id, v_pct, v_net) returning * into v_order; return v_order; end $$; -- --------------------------------------------------------------------- -- 5) التسليم (نسخة محدّثة): بيبدأ الاحتجاز ويسجل مستحقات البائع "معلّقة" -- --------------------------------------------------------------------- create or replace function public.deliver_order(p_order_id bigint, p_secret text default null) returns void language plpgsql security definer set search_path = public as $$ declare v_o public.orders; v_secret text; v_warranty timestamptz; v_hold timestamptz; begin perform public._require_admin(); select * into v_o from public.orders where id = p_order_id for update; if not found then raise exception 'order_not_found'; end if; if v_o.status not in ('paid', 'problem') then raise exception 'bad_status'; end if; if v_o.inventory_item_id is not null then select secret into v_secret from public.inventory_items where id = v_o.inventory_item_id; update public.inventory_items set status = 'sold' where id = v_o.inventory_item_id; end if; v_secret := coalesce(nullif(trim(p_secret), ''), v_secret); if v_secret is null then raise exception 'secret_required'; end if; insert into public.order_secrets (order_id, secret) values (v_o.id, v_secret) on conflict (order_id) do update set secret = excluded.secret, created_at = now(); v_warranty := now() + make_interval(hours => v_o.warranty_hours); v_hold := now() + make_interval(hours => public._setting('hold_hours', '72')::int); update public.orders set status = 'delivered', delivered_at = now(), warranty_until = v_warranty, release_at = greatest(v_warranty, v_hold), updated_at = now() where id = v_o.id; if v_o.seller_id is not null and not exists (select 1 from public.wallet_entries where order_id = v_o.id and kind = 'sale') then insert into public.wallet_entries (seller_id, order_id, kind, pending_delta, note) values (v_o.seller_id, v_o.id, 'sale', v_o.seller_net, 'بيع ' || v_o.code); insert into public.transactions (order_id, user_id, type, amount_egp, note) values (v_o.id, v_o.seller_id, 'commission', v_o.price_egp - v_o.seller_net, 'عمولة المنصة ' || v_o.commission_pct::text || '%'); end if; end $$; -- --------------------------------------------------------------------- -- 6) حل المشاكل (نسخة محدّثة): الاسترجاع بيخصم من المعلّق عند البائع -- --------------------------------------------------------------------- create or replace function public.resolve_order(p_order_id bigint, p_action text, p_note text default null) returns void language plpgsql security definer set search_path = public as $$ declare v_o public.orders; begin perform public._require_admin(); select * into v_o from public.orders where id = p_order_id for update; if not found then raise exception 'order_not_found'; end if; if p_action = 'refund' then if v_o.status not in ('paid', 'delivered', 'problem') then raise exception 'bad_status'; end if; update public.orders set status = 'refunded', admin_note = left(p_note, 500), updated_at = now() where id = v_o.id; insert into public.transactions (order_id, user_id, type, amount_egp, note) values (v_o.id, v_o.buyer_id, 'refund', v_o.price_egp, left(p_note, 200)); update public.inventory_items set status = 'available', order_id = null where id = v_o.inventory_item_id and status = 'reserved'; if v_o.seller_id is not null and exists (select 1 from public.wallet_entries where order_id = v_o.id and kind = 'sale') and not exists (select 1 from public.wallet_entries where order_id = v_o.id and kind in ('release', 'refund_reversal')) then insert into public.wallet_entries (seller_id, order_id, kind, pending_delta, note) values (v_o.seller_id, v_o.id, 'refund_reversal', -v_o.seller_net, 'استرجاع للمشتري ' || v_o.code); insert into public.transactions (order_id, user_id, type, amount_egp, note) values (v_o.id, v_o.seller_id, 'commission', -(v_o.price_egp - v_o.seller_net), 'عكس العمولة بسبب الاسترجاع'); end if; elsif p_action = 'complete' then if v_o.status not in ('delivered', 'problem') then raise exception 'bad_status'; end if; update public.orders set status = 'completed', admin_note = left(p_note, 500), updated_at = now() where id = v_o.id; elsif p_action = 'cancel' then if v_o.status not in ('awaiting_payment', 'under_review') then raise exception 'bad_status'; end if; update public.orders set status = 'cancelled', admin_note = left(p_note, 500), updated_at = now() where id = v_o.id; else raise exception 'bad_action'; end if; end $$; -- --------------------------------------------------------------------- -- 7) تحرير المبالغ اللي خلصت فترة احتجازها (معلّق -> متاح) -- الطلبات في حالة "مشكلة" مش بتتحرر. بتتنادى تلقائي عند فتح المحفظة. -- --------------------------------------------------------------------- create or replace function public.release_matured_orders() returns int language plpgsql security definer set search_path = public as $$ declare r record; n int := 0; begin for r in select id, seller_id, seller_net, status from public.orders where status in ('delivered', 'completed') and released_at is null and release_at is not null and release_at <= now() for update skip locked loop if r.seller_id is not null and exists (select 1 from public.wallet_entries where order_id = r.id and kind = 'sale') then insert into public.wallet_entries (seller_id, order_id, kind, pending_delta, available_delta, note) values (r.seller_id, r.id, 'release', -r.seller_net, r.seller_net, 'انتهاء فترة الاحتجاز'); end if; update public.orders set released_at = now(), status = case when status = 'delivered' then 'completed' else status end, updated_at = now() where id = r.id; n := n + 1; end loop; return n; end $$; revoke all on function public.release_matured_orders() from public, anon; grant execute on function public.release_matured_orders() to authenticated; -- الأدمن يمدّد الاحتجاز على طلب معيّن (مثلًا للاشتباه) create or replace function public.extend_hold(p_order_id bigint, p_hours int) returns void language plpgsql security definer set search_path = public as $$ begin perform public._require_admin(); if p_hours is null or p_hours <= 0 or p_hours > 720 then raise exception 'bad_hours'; end if; update public.orders set release_at = coalesce(release_at, now()) + make_interval(hours => p_hours), updated_at = now() where id = p_order_id and released_at is null; if not found then raise exception 'bad_status'; end if; end $$; revoke all on function public.extend_hold(bigint, int) from public, anon; grant execute on function public.extend_hold(bigint, int) to authenticated; -- --------------------------------------------------------------------- -- 8) محفظة البائع + طلب السحب -- --------------------------------------------------------------------- create or replace function public.my_wallet() returns jsonb language plpgsql security definer set search_path = public as $$ declare v_uid uuid := auth.uid(); v_p numeric; v_a numeric; v_q numeric; begin if v_uid is null then raise exception 'login_required'; end if; perform public.release_matured_orders(); select coalesce(sum(pending_delta), 0), coalesce(sum(available_delta), 0) into v_p, v_a from public.wallet_entries where seller_id = v_uid; select coalesce(sum(amount_egp), 0) into v_q from public.payout_requests where seller_id = v_uid and status = 'requested'; return jsonb_build_object( 'pending', v_p, 'available', v_a, 'in_payout_queue', v_q, 'next_payout_date', public.next_payout_date(now()), 'hold_hours', public._setting('hold_hours', '72')::int, 'min_payout', public._setting('min_payout_egp', '100')::numeric ); end $$; create or replace function public.request_payout(p_amount numeric, p_method text, p_account text) returns public.payout_requests language plpgsql security definer set search_path = public as $$ declare v_uid uuid := auth.uid(); v_avail numeric; v_min numeric; v_amt numeric(10,2); v_row public.payout_requests; begin if v_uid is null then raise exception 'login_required'; end if; if not exists (select 1 from public.profiles where id = v_uid and role = 'seller') then raise exception 'not_seller'; end if; perform pg_advisory_xact_lock(hashtext('wallet:' || v_uid::text)); perform public.release_matured_orders(); v_min := public._setting('min_payout_egp', '100')::numeric; v_amt := round(p_amount, 2); if v_amt is null or v_amt < v_min then raise exception 'below_min_payout'; end if; select coalesce(sum(available_delta), 0) into v_avail from public.wallet_entries where seller_id = v_uid; if v_amt > v_avail then raise exception 'insufficient_balance'; end if; insert into public.payout_requests (seller_id, amount_egp, method, account, batch_date) values (v_uid, v_amt, p_method, trim(p_account), public.next_payout_date(now())) returning * into v_row; insert into public.wallet_entries (seller_id, payout_id, kind, available_delta, note) values (v_uid, v_row.id, 'payout_request', -v_amt, 'طلب سحب #' || v_row.id::text); return v_row; end $$; revoke all on function public.my_wallet() from public, anon; revoke all on function public.request_payout(numeric, text, text) from public, anon; grant execute on function public.my_wallet() to authenticated; grant execute on function public.request_payout(numeric, text, text) to authenticated; -- --------------------------------------------------------------------- -- 9) أزرار الأدمن: تم الدفع / رفض السحب / ملخص الالتزامات -- --------------------------------------------------------------------- create or replace function public.mark_payout_paid(p_id bigint, p_note text default null) returns void language plpgsql security definer set search_path = public as $$ declare v_r public.payout_requests; begin perform public._require_admin(); select * into v_r from public.payout_requests where id = p_id and status = 'requested' for update; if not found then raise exception 'bad_status'; end if; update public.payout_requests set status = 'paid', processed_at = now(), admin_note = left(p_note, 300) where id = p_id; insert into public.transactions (user_id, type, amount_egp, note) values (v_r.seller_id, 'payout', v_r.amount_egp, 'سحب #' || p_id::text || ' - ' || v_r.method); end $$; create or replace function public.reject_payout(p_id bigint, p_reason text default null) returns void language plpgsql security definer set search_path = public as $$ declare v_r public.payout_requests; begin perform public._require_admin(); select * into v_r from public.payout_requests where id = p_id and status = 'requested' for update; if not found then raise exception 'bad_status'; end if; update public.payout_requests set status = 'rejected', processed_at = now(), admin_note = left(p_reason, 300) where id = p_id; insert into public.wallet_entries (seller_id, payout_id, kind, available_delta, note) values (v_r.seller_id, p_id, 'payout_reversal', v_r.amount_egp, 'رفض طلب سحب #' || p_id::text); end $$; -- أرصدة كل البائعين create or replace function public.admin_wallets() returns table (uid uuid, name text, pending numeric, available numeric, queued numeric) language plpgsql security definer set search_path = public as $$ begin perform public._require_admin(); perform public.release_matured_orders(); return query select p.id, p.display_name, coalesce((select sum(w.pending_delta) from public.wallet_entries w where w.seller_id = p.id), 0), coalesce((select sum(w.available_delta) from public.wallet_entries w where w.seller_id = p.id), 0), coalesce((select sum(r.amount_egp) from public.payout_requests r where r.seller_id = p.id and r.status = 'requested'), 0) from public.profiles p where p.role = 'seller' order by p.display_name; end $$; -- إجمالي اللي عليك للبائعين (لازم تفضل فلوس بالقدر ده في محفظتك الفعلية) create or replace function public.admin_escrow_summary() returns jsonb language plpgsql security definer set search_path = public as $$ declare v_p numeric; v_a numeric; v_q numeric; begin perform public._require_admin(); perform public.release_matured_orders(); select coalesce(sum(pending_delta), 0), coalesce(sum(available_delta), 0) into v_p, v_a from public.wallet_entries; select coalesce(sum(amount_egp), 0) into v_q from public.payout_requests where status = 'requested'; return jsonb_build_object('pending', v_p, 'available', v_a, 'queued', v_q, 'owed_total', v_p + v_a + v_q); end $$; revoke all on function public.mark_payout_paid(bigint, text) from public, anon; revoke all on function public.reject_payout(bigint, text) from public, anon; revoke all on function public.admin_wallets() from public, anon; revoke all on function public.admin_escrow_summary() from public, anon; grant execute on function public.mark_payout_paid(bigint, text) to authenticated; grant execute on function public.reject_payout(bigint, text) to authenticated; grant execute on function public.admin_wallets() to authenticated; grant execute on function public.admin_escrow_summary() to authenticated; -- إشعار ديسكورد عند طلب سحب جديد create or replace function public.trg_payout_notify() returns trigger language plpgsql security definer set search_path = public as $$ begin perform public.notify_admin('💰 طلب سحب جديد #' || new.id::text || ' — ' || new.amount_egp::text || ' EGP — دفعة يوم ' || new.batch_date::text);
